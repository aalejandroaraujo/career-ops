% This is a placeholder template. The AI agent fills every double-brace token
% from cv.md / config/profile.yml at generation time, writes the filled .tex,
% then validates + compiles it with generate-latex.mjs.
%
% Compile: pdfLaTeX (what generate-latex.mjs picks for this template — the
% \else branch below uses the bundled Type1 lato package, so it needs no system
% font) or XeLaTeX/LuaLaTeX, which take the \iftutex branch and require Lato
% installed as a system font.
% The academic bibliography (biber) has been intentionally removed — a job CV
% doesn't need it, and it keeps the toolchain to a single engine.
%------------------------

\documentclass[10pt,a4paper,withhyper]{altacv}

% PDF metadata: the candidate, not the toolchain. Without this, Creator and
% Producer read "LaTeX with hyperref" / "pdfTeX-1.40" in the recruiter's viewer.
% Filled from the payload name and tagline; cv-verify.mjs asserts it.
\hypersetup{unicode,pdfauthor={Test Candidate},pdfcreator={Test Candidate},pdfproducer={Test Candidate},pdftitle={Test Candidate | Senior Backend Engineer}}

% Page layout
\geometry{left=1cm,right=1cm,top=1.5cm,bottom=1.5cm,columnsep=1cm}

% paracol drives the two-column body
\usepackage{paracol}

% Font. Under XeLaTeX/LuaLaTeX use the system Lato; under pdfLaTeX use the
% bundled lato package (Type1, no system font needed). generate-latex.mjs
% ensures the right one is available for whichever engine it picks.
\iftutex
  \setmainfont{Lato}
\else
  \usepackage[default]{lato}
\fi

% Body leading. The experience column is packed to the 2-page limit, so a global
% leading bump spills a 3rd page — kept neutral. (Breathing room is added in the
% shorter right column instead; see \cvcert spacing below.)
\linespread{1.0}

% Colours — accent drives headings/marks. Tweak to taste.
\definecolor{Blue}{HTML}{0080ff}
\definecolor{SlateGrey}{HTML}{2E2E2E}
\definecolor{LightGrey}{HTML}{666666}
\definecolor{Black}{HTML}{000000}
\colorlet{heading}{Black}
\colorlet{headingrule}{Black}
\colorlet{accent}{Blue}
% Body text is BLACK, not AltaCV's default #666666 grey. The experience bullets
% render at \small in Lato, and at that size/weight anything lighter than pure
% black reads as grey next to the black headings and role titles (#2E2E2E was
% tried first and still looked washed out). Hierarchy comes from size and weight
% here, not from colour. `body!30` (dividers, unfilled rating markers) stays a
% light grey, which is AltaCV's intent for them.
\colorlet{emphasis}{Black}
\colorlet{body}{Black}

% Markers
\renewcommand{\cvItemMarker}{{\small\textbullet}}
\renewcommand{\cvRatingMarker}{\faCircle}

% Compact one-line certification entry with a small marker — lets many certs fit
% without the tall two-row \cvachievement layout (whose icon is \LARGE).
\newcommand{\cvcert}[1]{%
  {\footnotesize\color{accent}\BeginAccSupp{method=escape,ActualText={}}\faCertificate\EndAccSupp{}}\hspace{0.6em}{\small #1}\par\medskip%
}

% Language/skill rating: circles PLUS the numeric level after them, so the level
% is unambiguous at a glance without counting dots. Whole row is \small and the
% circles \footnotesize so a long label (e.g. "German (B1, B2 in progress)")
% still fits on ONE line in the narrow right column.
\newcommand{\cvskillnum}[2]{%
  {\small\textcolor{emphasis}{\textbf{#1}}}\hfill
  {\footnotesize
  \foreach \lvl in {1,...,5}{%
    \ifdimgreater{\lvl bp}{#2 bp}{\color{body!30}}{\color{accent}}\cvRatingMarker%
  }}\hspace{0.4em}{\small\bfseries\color{accent}#2}\par%
}

% Mirror the photo horizontally (subject faces into the text) while KEEPING the
% circular crop, AND leaving a little headroom above the head. Instead of the
% centered overzoom fill (which crops right into the top of the head), we clip a
% circle and drop the image inside it, shifted DOWN by \photoheadroom — so the
% top of the circle shows a gap above the head. \reflectbox mirrors horizontally;
% vertical framing is unaffected. width=1.04\linewidth guarantees the circle stays
% fully covered after the shift (portrait source, so height already overflows).
\newlength{\photoheadroom}
\setlength{\photoheadroom}{0.34cm}% ~7.5% of the 4.4cm circle: gap above the head
\makeatletter
\renewcommand{\@makeaphoto}[2]{%
  \begin{minipage}{#1}%
    \if@normalphoto
      \reflectbox{\includegraphics[width=\linewidth]{#2}}%
    \else
      \reflectbox{%
        \begin{tikzpicture}
          \clip (0,0) circle (0.5\linewidth);
          \node[inner sep=0pt] at (0,-\photoheadroom)
            {\includegraphics[width=1.04\linewidth]{#2}};
        \end{tikzpicture}%
      }%
    \fi%
  \end{minipage}%
}
% Nudge the right-hand photo inward from the page edge (moves it toward the left).
% Vertical position of the right-hand photo. \raisebox with a negative value
% lowers the circle down the page. 6mm is the practical maximum for a 2-page CV:
% the header grows with it, and 7mm pushes the body onto a third page.
\newlength{\photodrop}
\setlength{\photodrop}{6mm}
\renewcommand{\altacv@add@photo@right}[1]{%
  % \hspace* (not \hspace): the photos are followed by \par, and TeX discards
  % trailing glue at a paragraph end, so a plain \hspace left the photo flush on
  % the 1cm page margin. cv-verify.mjs asserts the margin after every build.
  \appto{\altacv@right@photos}{\raisebox{-\photodrop}{\@makeaphoto{\altacv@photo@diam@right}{#1}}\hspace*{1.2cm}}%
  \addtolength{\altacv@photos@width}{\altacv@photo@diam@right}%
  \addtolength{\altacv@photos@width}{1.2cm}%
}
% Bigger personal-details text (was \footnotesize) so the contact block has more body.
\renewcommand{\personalinfofont}{\small\bfseries}
% Header override: distribute name / tagline / personal-info EVENLY across the full
% photo height (via a \vbox to the photo diameter with \vfill between the blocks),
% so the contact details spread down to fill the blank gap above the Summary instead
% of clustering under the tagline. Header height stays bounded by the photo, so this
% does not push the body down / add a page.
\renewcommand{\makecvheader}{%
  \begingroup
    \altacv@left@photos\hfill%
    \begin{minipage}[c]{\dimexpr\linewidth-\altacv@photos@width}%
      \raggedright
      \vbox to \altacv@photo@diam@right{%
        \raggedright
        {\namefont\color{name}\MakeUppercase{\@name}\par}%
        \vfill
        {\taglinefont\color{tagline}\@tagline\par}%
        \vfill
        {\personalinfofont\@personalinfo\par}%
        \vfill
      }%
    \end{minipage}\hfill%
    \altacv@right@photos\par%
  \endgroup\medskip
}
\makeatother
